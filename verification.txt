def PolynomialVisibility.HasTwoDistinctRoots : Polynomial ℤ → Prop :=
fun F => ∃ z w, z ≠ w ∧ (Polynomial.aeval z) F = 0 ∧ (Polynomial.aeval w) F = 0
def PolynomialVisibility.Visible : Polynomial ℤ → ℕ → ℕ → Prop :=
fun F a h =>
  0 < a ∧
    0 < h ∧
      ∃ t,
        ↑h = t * ↑(Polynomial.eval (↑a) F) ∧
          ∀ (u : ℕ), 0 < u → PolynomialVisibility.IsPositiveInteger (t * ↑(Polynomial.eval (↑u) F)) → a ≤ u
def PolynomialVisibility.HasVisibilityDensityOne : Polynomial ℤ → Prop :=
fun F => Filter.Tendsto (fun N => ↑(PolynomialVisibility.visibleCount F N) / ↑N ^ 2) Filter.atTop (nhds 1)
